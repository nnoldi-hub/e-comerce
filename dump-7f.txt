<?php

namespace Tests\Feature\Pricing;

use App\Domain\Checkout\DTOs\PlaceOrderData;
use App\Domain\Pricing\DTOs\CartTotals;
use App\Domain\Pricing\DTOs\PricingInput;
use App\Domain\Pricing\PriceCalculator;
use App\Models\Cart;
use App\Models\CartItem;
use App\Models\Coupon;
use App\Models\Product;
use App\Models\ProductVariant;
use App\Models\ShippingMethod;
use App\Models\ShippingRate;
use App\Models\ShippingZone;
use App\Models\TaxRate;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Illuminate\Support\Str;
use RuntimeException;
use Tests\TestCase;

abstract class PricingTestCase extends TestCase
{
    use RefreshDatabase;

    /**
     * RefreshDatabase rulează migrate:fresh. Refuzăm să pornim dacă baza
     * de test nu se termină în "_test", ca să nu ștergem baza de dezvoltare.
     */
    protected function beforeRefreshingDatabase(): void
    {
        $connection = config('database.default');
        $database = (string) config("database.connections.{$connection}.database");

        if (! str_ends_with($database, '_test')) {
            throw new RuntimeException(
                "Baza '{$database}' nu pare a fi de test. Setează în phpunit.xml DB_CONNECTION=mysql și DB_DATABASE=e_comerce_test."
            );
        }
    }

    /**
     * @param  array<string, mixed>  $variant  suprascrieri pentru variantă
     * @param  array<string, mixed>  $product  suprascrieri pentru produs
     */
    protected function makeVariant(int $price, int $stock = 100, array $variant = [], array $product = []): ProductVariant
    {
        $productModel = Product::forceCreate(array_merge([
            'ulid' => (string) Str::ulid(),
            'name' => 'Produs '.Str::random(6),
            'slug' => 'produs-'.Str::lower(Str::random(10)),
            'status' => 'active',
            'published_at' => now()->subDay(),
        ], $product));

        return ProductVariant::forceCreate(array_merge([
            'product_id' => $productModel->id,
            'sku' => 'SKU-'.Str::upper(Str::random(8)),
            'price' => $price,
            'stock_on_hand' => $stock,
        ], $variant));
    }

    /**
     * @param  list<array{0: ProductVariant, 1: int}>  $lines
     */
    protected function makeCart(array $lines): Cart
    {
        $cart = Cart::create(['currency' => 'RON']);

        foreach ($lines as [$variant, $quantity]) {
            CartItem::create([
                'cart_id' => $cart->id,
                'product_variant_id' => $variant->id,
                'quantity' => $quantity,
                'unit_price_snapshot' => 1, // depășit intenționat
            ]);
        }

        return $cart;
    }

    protected function makeTax(int $bps, bool $inclusive = true, string $country = 'RO'): TaxRate
    {
        return TaxRate::forceCreate([
            'name' => "TVA {$bps}",
            'rate_bps' => $bps,
            'country' => $country,
            'is_inclusive' => $inclusive,
            'is_default' => true,
            'is_active' => true,
        ]);
    }

    /**
     * @param  array<string, mixed>  $attributes
     */
    protected function makeCoupon(array $attributes = []): Coupon
    {
        return Coupon::forceCreate(array_merge([
            'code' => 'T'.Str::upper(Str::random(7)),
            'type' => 'percent',
            'value' => 1000,
            'is_active' => true,
        ], $attributes));
    }

    /** Curier în RO: 0–19999 => 1999; de la 20000 => gratuit. */
    protected function makeCourier(): ShippingMethod
    {
        $zone = ShippingZone::forceCreate(['name' => 'România', 'countries' => ['RO'], 'is_active' => true]);
        $method = ShippingMethod::forceCreate(['shipping_zone_id' => $zone->id, 'name' => 'Curier', 'is_active' => true]);

        ShippingRate::forceCreate(['shipping_method_id' => $method->id, 'min_subtotal' => 0, 'max_subtotal' => 19999, 'price' => 1999]);
        ShippingRate::forceCreate(['shipping_method_id' => $method->id, 'min_subtotal' => 20000, 'max_subtotal' => null, 'price' => 0]);

        return $method;
    }

    /**
     * @param  array<string, mixed>  $overrides
     */
    protected function makeOrderData(int $shippingMethodId, array $overrides = []): PlaceOrderData
    {
        $address = [
            'first_name' => 'Ion',
            'last_name' => 'Popescu',
            'phone' => '0712345678',
            'address_line1' => 'Str. Exemplu 1',
            'city' => 'București',
            'postal_code' => '010101',
            'country' => 'RO',
        ];

        return new PlaceOrderData(...array_merge([
            'email' => 'client@example.com',
            'billingAddress' => $address,
            'shippingAddress' => $address,
            'shippingMethodId' => $shippingMethodId,
            'paymentMethod' => 'card',
            'idempotencyKey' => (string) Str::uuid(),
        ], $overrides));
    }

    protected function calculate(Cart $cart, ?PricingInput $input = null): CartTotals
    {
        return app(PriceCalculator::class)->calculate($cart, $input);
    }
}

<?php

namespace Tests\Feature\Payments;

use App\Domain\Checkout\Actions\PlaceOrderAction;
use App\Models\Order;
use App\Models\Payment;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Mail;
use Illuminate\Testing\TestResponse;
use Tests\Feature\Pricing\PricingTestCase;

/**
 * Bază pentru teste care pornesc de la o comandă cu Stripe deja plătită prin
 * webhook (cu PaymentIntent `pi_123`).
 *
 * Http::fake adaugă stub-uri la cele existente și câștigă primul potrivit, deci
 * răspunsul pentru rambursări se dă o singură dată, la crearea comenzii.
 */
abstract class StripePaidOrderTestCase extends PricingTestCase
{
    protected const SECRET = 'whsec_test';

    protected function setUp(): void
    {
        parent::setUp();

        Mail::fake();

        config([
            'shop.payments.methods.card' => 'stripe',
            'shop.payments.gateways.stripe.secret' => 'sk_test_123',
            'shop.payments.gateways.stripe.webhook_secret' => self::SECRET,
        ]);
    }

    /**
     * @return array{0: Order, 1: Payment}
     */
    protected function paidStripeOrder(mixed $refundStub = null, mixed $refundGetStub = null): array
    {
        Http::fake([
            'api.stripe.com/v1/checkout/sessions/*' => Http::response(['id' => 'cs_test_123', 'payment_intent' => 'pi_123']),
            'api.stripe.com/v1/checkout/sessions' => Http::response(['id' => 'cs_test_123', 'url' => 'https://checkout.stripe.test/cs_test_123']),
            'api.stripe.com/v1/refunds/*' => $refundGetStub ?? Http::response(['id' => 're_p', 'status' => 'pending']),
            'api.stripe.com/v1/refunds' => $refundStub ?? Http::response(['id' => 're_p', 'status' => 'pending']),
        ]);

        $courier = $this->makeCourier();
        $variant = $this->makeVariant(12100, stock: 10);
        $cart = $this->makeCart([[$variant, 2]]);

        $order = $this->app->make(PlaceOrderAction::class)->execute(
            $cart,
            $this->makeOrderData($courier->id, ['paymentMethod' => 'card']),
        );

        $payment = Payment::query()->where('order_id', $order->id)->firstOrFail();

        $this->webhook('checkout.session.completed', [
            'id' => 'cs_test_123',
            'amount_total' => $payment->amount->amount(),
            'currency' => 'ron',
            'payment_status' => 'paid',
            'payment_intent' => 'pi_123',
        ], 'evt_paid')->assertOk();

        return [$order->refresh(), $payment->refresh()];
    }

    /**
     * @param  array<string, mixed>  $object
     */
    protected function webhook(string $type, array $object, string $eventId): TestResponse
    {
        $payload = (string) json_encode([
            'id' => $eventId,
            'type' => $type,
            'data' => ['object' => $object],
        ]);

        $timestamp = (string) time();
        $signature = hash_hmac('sha256', $timestamp.'.'.$payload, self::SECRET);

        return $this->call('POST', '/webhooks/payments/stripe', [], [], [], [
            'HTTP_STRIPE_SIGNATURE' => "t={$timestamp},v1={$signature}",
            'CONTENT_TYPE' => 'application/json',
            'HTTP_ACCEPT' => 'application/json',
        ], $payload);
    }

    /**
     * @param  array<string, mixed>  $extra
     * @return array<string, mixed>
     */
    protected function refundObject(string $id, string $status, int $amount, array $extra = []): array
    {
        return array_merge([
            'id' => $id,
            'status' => $status,
            'amount' => $amount,
            'currency' => 'ron',
            'payment_intent' => 'pi_123',
        ], $extra);
    }
}

            'first_name' => 'Ion',
            'last_name' => 'Popescu',

            'address_line1' => 'Str. Exemplu 1',
            'city' => 'București',
            'postal_code' => '010101',
            'country' => 'ro',
            'shipping_method_id' => $shippingMethodId,

            'idempotency_key' => (string) Str::uuid(),
            'terms' => '1',

    public function test_a_guest_can_place_an_order(): void
    {
        $courier = $this->makeCourier();
        $cart = $this->makeCart([[$this->makeVariant(12100, stock: 5), 1]]);

        $this->withSession(['cart_ulid' => $cart->ulid])
            ->post(route('checkout.store'), $this->payload($courier->id))
            ->assertRedirect(route('checkout.success'))
            ->assertSessionHas('order_number');

        $order = Order::query()->firstOrFail();
        $this->assertSame('client@example.com', $order->email);
        $this->assertSame('RO', $order->shipping_address['country']);
        $this->assertNull($order->user_id);
    }

    public function test_required_fields_are_validated(): void
    {
        $this->post(route('checkout.store'), [])->assertSessionHasErrors([
            'email', 'first_name', 'last_name', 'phone', 'address_line1', 'city',
            'postal_code', 'country', 'shipping_method_id', 'payment_method',
            'idempotency_key', 'terms',
        ]);

        $this->assertSame(0, Order::query()->count());
    }

    public function test_a_company_needs_a_valid_cui(): void
    {
        $courier = $this->makeCourier();
        $cart = $this->makeCart([[$this->makeVariant(5000), 1]]);
        $session = ['cart_ulid' => $cart->ulid];

        $this->withSession($session)
            ->post(route('checkout.store'), $this->payload($courier->id, [
                'is_company' => '1', 'company_name' => 'Firma SRL', 'cui' => '1234566',

        $this->assertSame('RO1234565', $order->billing_address['cui']);

    {
        $courier = $this->makeCourier();
        $cart = $this->makeCart([[$this->makeVariant(5000), 1]]);

        $this->withSession(['cart_ulid' => $cart->ulid])

    }

<?php

namespace Tests\Feature\Payments;

use App\Domain\Payments\Actions\RefundPaymentAction;
use App\Enums\OrderStatus;
use App\Enums\PaymentStatus;
use App\Enums\RefundStatus;
use App\Models\Payment;
use App\Models\Refund;
use App\Models\WebhookEvent;
use Illuminate\Support\Facades\Http;

class RefundWebhookTest extends StripePaidOrderTestCase
{
    public function test_the_payment_intent_is_stored_when_the_payment_is_confirmed(): void
    {
        [, $payment] = $this->paidStripeOrder();

        $this->assertSame('pi_123', $payment->provider_intent);
    }

    public function test_a_full_refund_made_in_the_stripe_dashboard_is_recorded(): void
    {
        [$order, $payment] = $this->paidStripeOrder();

        $this->webhook('refund.created', $this->refundObject('re_ext', 'succeeded', $payment->amount->amount()), 'evt_1')
            ->assertOk();

        $refund = Refund::query()->firstOrFail();
        $this->assertSame(RefundStatus::SUCCEEDED, $refund->status);
        $this->assertSame('re_ext', $refund->provider_reference);
        $this->assertSame($payment->amount->amount(), $refund->amount->amount());
        $this->assertNull($refund->created_by);
        $this->assertStringContainsString('procesatorului', (string) $refund->reason);

        $this->assertSame(PaymentStatus::REFUNDED, $payment->fresh()->status);
        $this->assertSame(PaymentStatus::REFUNDED, $order->fresh()->payment_status);
        $this->assertSame(OrderStatus::REFUNDED, $order->fresh()->status);
    }

    public function test_a_partial_refund_made_in_the_dashboard_keeps_the_order_open(): void
    {
        [$order, $payment] = $this->paidStripeOrder();

        $this->webhook('refund.created', $this->refundObject('re_ext', 'succeeded', 1000), 'evt_1')->assertOk();

        $this->assertSame(PaymentStatus::PARTIALLY_REFUNDED, $payment->fresh()->status);
        $this->assertSame(OrderStatus::PAID, $order->fresh()->status);
    }

    public function test_a_pending_refund_of_ours_is_completed_by_the_webhook_without_duplicates(): void
    {
        [, $payment] = $this->paidStripeOrder();

        $local = $this->app->make(RefundPaymentAction::class)->execute($payment, 1500, 'test');

        $this->assertSame(RefundStatus::PENDING, $local->status);
        $this->assertSame('re_p', $local->provider_reference);

        $this->webhook('refund.updated', $this->refundObject('re_p', 'succeeded', 1500, [
            'metadata' => ['idempotency_key' => 'ecom-refund-'.$local->id],
        ]), 'evt_2')->assertOk();

        $this->assertSame(1, Refund::query()->count());

        $refund = $local->fresh();
        $this->assertSame(RefundStatus::SUCCEEDED, $refund->status);
        $this->assertNotNull($refund->processed_at);
        $this->assertSame('test', $refund->reason);
        $this->assertSame(PaymentStatus::PARTIALLY_REFUNDED, $payment->fresh()->status);
    }

    public function test_a_webhook_that_beats_the_saved_reference_adopts_the_local_refund(): void
    {
        [, $payment] = $this->paidStripeOrder();

        // Faza 1 s-a încheiat, dar referința nu a apucat să fie salvată.
        $local = Refund::create([
            'payment_id' => $payment->id,
            'order_id' => $payment->order_id,
            'amount' => 1500,
            'currency' => $payment->currency,
            'status' => RefundStatus::PENDING,
        ]);

        $this->webhook('refund.created', $this->refundObject('re_early', 'succeeded', 1500, [
            'metadata' => ['idempotency_key' => 'ecom-refund-'.$local->id],
        ]), 'evt_3')->assertOk();

        $this->assertSame(1, Refund::query()->count());
        $this->assertSame('re_early', $local->fresh()->provider_reference);
        $this->assertSame(RefundStatus::SUCCEEDED, $local->fresh()->status);
    }

    public function test_the_same_event_is_processed_once_and_a_finished_refund_is_not_reopened(): void
    {
        [, $payment] = $this->paidStripeOrder();

        $object = $this->refundObject('re_ext', 'succeeded', 1000);

        $this->webhook('refund.created', $object, 'evt_1')->assertOk();
        $this->webhook('refund.created', $object, 'evt_1')->assertOk();

        // Un alt eveniment pentru aceeași rambursare, deja încheiată.
        $this->webhook('refund.updated', $this->refundObject('re_ext', 'failed', 1000), 'evt_2')->assertOk();

        $this->assertSame(1, Refund::query()->count());
        $this->assertSame(RefundStatus::SUCCEEDED, Refund::query()->firstOrFail()->status);
        $this->assertSame(PaymentStatus::PARTIALLY_REFUNDED, $payment->fresh()->status);
    }

    public function test_a_failed_refund_does_not_touch_the_payment(): void
    {
        [, $payment] = $this->paidStripeOrder();

        $local = $this->app->make(RefundPaymentAction::class)->execute($payment, 1500);

        $this->webhook('refund.failed', $this->refundObject('re_p', 'failed', 1500, [
            'metadata' => ['idempotency_key' => 'ecom-refund-'.$local->id],
        ]), 'evt_4')->assertOk();

        $this->assertSame(RefundStatus::FAILED, $local->fresh()->status);
        $this->assertSame(PaymentStatus::PAID, $payment->fresh()->status);
    }

    public function test_a_payment_confirmed_before_the_intent_column_is_found_through_its_payload(): void
    {
        [, $payment] = $this->paidStripeOrder();

        Payment::query()->whereKey($payment->id)->update(['provider_intent' => null]);

        $this->webhook('refund.created', $this->refundObject('re_old', 'succeeded', 1000), 'evt_5')->assertOk();

        $this->assertSame(1, Refund::query()->count());
        $this->assertSame(PaymentStatus::PARTIALLY_REFUNDED, $payment->fresh()->status);
    }

    public function test_a_refund_for_an_unknown_payment_is_retried_then_dropped(): void
    {
        $this->paidStripeOrder();

        $object = $this->refundObject('re_x', 'succeeded', 1000, ['payment_intent' => 'pi_unknown']);

        $this->webhook('refund.created', $object, 'evt_6')->assertStatus(409);

        $this->travel(11)->minutes();

        $this->webhook('refund.created', $object, 'evt_6')->assertStatus(202);

        $this->assertSame(0, Refund::query()->count());
        $this->assertSame('failed', WebhookEvent::query()->where('event_id', 'evt_6')->firstOrFail()->status);
    }

    public function test_aggregate_charge_events_are_ignored(): void
    {
        $this->paidStripeOrder();

        $this->webhook('charge.refunded', ['id' => 'ch_1', 'payment_intent' => 'pi_123'], 'evt_7')
            ->assertStatus(202)
            ->assertJson(['status' => 'ignored']);

        $this->assertSame(0, Refund::query()->count());
    }

    public function test_the_stripe_refund_request_carries_the_local_key_in_its_metadata(): void
    {
        [, $payment] = $this->paidStripeOrder();

        $local = $this->app->make(RefundPaymentAction::class)->execute($payment, 1500);

        Http::assertSent(fn ($request) => $request->url() === 'https://api.stripe.com/v1/refunds'
            && ($request['metadata']['idempotency_key'] ?? null) === 'ecom-refund-'.$local->id);
    }
}

===== app/Domain/Shipping/Actions/MarkShipmentDeliveredAction.php =====

<?php

namespace App\Domain\Shipping\Actions;

use App\Domain\Orders\Actions\TransitionOrderStatusAction;
use App\Domain\Orders\Exceptions\InvalidOrderTransitionException;
use App\Domain\Shipping\Exceptions\ShipmentException;
use App\Enums\OrderStatus;
use App\Enums\PaymentStatus;
use App\Enums\ShipmentStatus;
use App\Models\Order;
use App\Models\Payment;
use App\Models\Shipment;
use Illuminate\Support\Facades\DB;

/**
 * Marchează o expediere ca livrată: comanda trece în `delivered`, iar la
 * ramburs plata se consideră încasată în momentul livrării (curierul a luat
 * banii), ca rambursările ulterioare să poată porni din `paid`.
 */
class MarkShipmentDeliveredAction
{
    public function __construct(private readonly TransitionOrderStatusAction $transitions) {}

    /**
     * @throws ShipmentException comanda nu este expediată
     * @throws InvalidOrderTransitionException tranziția comenzii a fost refuzată
     */
    public function execute(Shipment $shipment, ?int $changedBy = null): Shipment
    {
        return DB::transaction(function () use ($shipment, $changedBy): Shipment {
            $locked = Shipment::query()->lockForUpdate()->findOrFail($shipment->id);

            if ($locked->status === ShipmentStatus::DELIVERED) {
                return $locked;
            }

            $order = Order::query()->lockForUpdate()->findOrFail($locked->order_id);

            if ($order->status !== OrderStatus::SHIPPED) {
                throw new ShipmentException(
                    "Comanda {$order->number} nu este în statusul „Expediată”.",
                    'not_shipped',
                );
            }

            $locked->update([
                'status' => ShipmentStatus::DELIVERED,
                'delivered_at' => now(),
            ]);

            $this->collectCashOnDelivery($order);

            $this->transitions->execute($order, OrderStatus::DELIVERED, $changedBy, 'Livrată');

            return $locked->refresh();
        });
    }

    private function collectCashOnDelivery(Order $order): void
    {
        if ($order->payment_method !== 'cod' || $order->payment_status !== PaymentStatus::PENDING) {
            return;
        }

        $collected = Payment::query()
            ->where('order_id', $order->id)
            ->where('provider', 'cod')
            ->where('status', PaymentStatus::PENDING)
            ->update(['status' => PaymentStatus::PAID, 'paid_at' => now()]);

        if ($collected > 0) {
            $order->update(['payment_status' => PaymentStatus::PAID]);
        }
    }
}

===== app/Http/Controllers/Payments/PaymentWebhookController.php =====

<?php

namespace App\Http\Controllers\Payments;

use App\Domain\Payments\Actions\ProcessPaymentWebhookAction;
use App\Domain\Payments\Exceptions\PaymentNotFoundException;
use App\Domain\Payments\Exceptions\UnsupportedGatewayException;
use App\Domain\Payments\PaymentGatewayManager;
use App\Http\Controllers\Controller;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Log;

class PaymentWebhookController extends Controller
{
    public function __invoke(
        Request $request,
        string $gateway,
        PaymentGatewayManager $gateways,
        ProcessPaymentWebhookAction $process,
    ): JsonResponse {
        try {
            $provider = $gateways->gateway($gateway);
        } catch (UnsupportedGatewayException) {
            abort(404);
        }

        if (! $provider->verifyWebhook($request)) {
            Log::warning('Webhook de plată respins: semnătură invalidă.', [
                'gateway' => $gateway,
                'ip' => $request->ip(),
            ]);

            return response()->json(['status' => 'invalid_signature'], 403);
        }

        $notification = $provider->parseWebhook($request);

        if ($notification === null) {
            return response()->json(['status' => 'ignored'], 202);
        }

        try {
            $outcome = $process->execute($notification);
        } catch (PaymentNotFoundException $exception) {
            // Probabil o cursă: providerul reîncearcă în următoarele minute.
            return response()->json(['status' => 'retry', 'message' => $exception->getMessage()], 409);
        }

        if ($outcome === ProcessPaymentWebhookAction::REJECTED) {
            // Eroare permanentă, deja înregistrată: 202 oprește retrimiterile.
            return response()->json(['status' => 'rejected'], 202);
        }

        return response()->json(['status' => 'ok']);
    }
}
